Yes — that’s actually a very realistic UX pattern used in event/ticketing apps 🌷

Since your app probably redirects users to:

* BookMyShow
* Eventbrite
* Meetup
* external booking pages

your app cannot directly know if booking succeeded unless you own the payment system.

So the standard industry approach is:

```text id="jlwmel"
User clicks booking link
   ↓
External website opens
   ↓
User comes back to app
   ↓
App asks:
"Did you complete your booking?"
   ↓
If YES:
Add event to Registered Events
Schedule reminder
Show success state
```

This is the cleanest solution for your architecture.

import 'package:shared_preferences/shared_preferences.dart';

class BookingService {

  static const String registeredEventsKey =
      'registered_events';

  static Future<void> registerEvent(
    String eventId,
  ) async {

    final prefs =
        await SharedPreferences.getInstance();

    final registeredEvents =
        prefs.getStringList(
              registeredEventsKey,
            ) ??
            [];

    if (!registeredEvents.contains(eventId)) {

      registeredEvents.add(eventId);

      await prefs.setStringList(
        registeredEventsKey,
        registeredEvents,
      );
    }
  }

  static Future<List<String>>
      getRegisteredEvents() async {

    final prefs =
        await SharedPreferences.getInstance();

    return prefs.getStringList(
          registeredEventsKey,
        ) ??
        [];
  }

  static Future<bool> isRegistered(
    String eventId,
  ) async {

    final prefs =
        await SharedPreferences.getInstance();

    final registeredEvents =
        prefs.getStringList(
              registeredEventsKey,
            ) ??
            [];

    return registeredEvents.contains(eventId);
  }
}

class EventDetailsScreen extends StatefulWidget {

  State<EventDetailsScreen> createState() =>

class _EventDetailsScreenState
    extends State<EventDetailsScreen>
    with WidgetsBindingObserver {

  bool openedBookingLink = false;

  bool isRegistered = false;

  @override
  void initState() {
    super.initState();

    WidgetsBinding.instance.addObserver(this);

    checkRegistrationStatus();
  }

  Future<void>
      checkRegistrationStatus() async {

    final registered =
        await BookingService.isRegistered(
      widget.event.id,
    );

    setState(() {
      isRegistered = registered;
    });
  }

  @override
  void dispose() {

    WidgetsBinding.instance.removeObserver(this);

    super.dispose();
  }

  @override
  void didChangeAppLifecycleState(
      AppLifecycleState state) {

    if (state == AppLifecycleState.resumed &&
        openedBookingLink) {

      showBookingConfirmationDialog();
    }
  }

  Future<void> openBookingLink() async {

    openedBookingLink = true;

    final Uri url =
        Uri.parse(widget.event.bookingUrl);

    await launchUrl(
      url,
      mode: LaunchMode.externalApplication,
    );
  }

  Future<void>
      showBookingConfirmationDialog() async {

    final bool? confirmed =
        await showDialog<bool>(

      context: context,

      builder: (context) {

        return AlertDialog(

          title: const Text(
            'Booking Confirmation',
          ),

          content: Text(
            'Did you complete booking for '
            '${widget.event.title}?',
          ),

          actions: [

            TextButton(
              onPressed: () {
                Navigator.pop(context, false);
              },
              child: const Text('Not Yet'),
            ),

            ElevatedButton(
              onPressed: () {
                Navigator.pop(context, true);
              },
              child: const Text('Yes'),
            ),
          ],
        );
      },
    );

    if (confirmed == true) {

      await BookingService.registerEvent(
        widget.event.id,
      );

      setState(() {
        isRegistered = true;
      });

      if (mounted) {

        ScaffoldMessenger.of(context)
            .showSnackBar(

          const SnackBar(
            content: Text(
              'Event added to Registered Events',

    openedBookingLink = false;
  }

    return Scaffold(

      appBar: AppBar(
        title: Text(widget.event.title),
      ),

      _RegisteredEventsScreenState();
